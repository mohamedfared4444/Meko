   163→   163→        title: String(payload.title || "فيديو برمجي — رندر كامل"),
   164→   164→        width: Number(process.env.INPUT_WIDTH) || Number(payload.width) || DEFAULT_WIDTH,
   165→   165→        height: Number(process.env.INPUT_HEIGHT) || Number(payload.height) || DEFAULT_HEIGHT,
   166→   166→        fps: Number(process.env.INPUT_FPS) || Number(payload.fps) || DEFAULT_FPS,

   171→   171→    },

   202→   202→
   203→   203→async function renderVideo({ html, width, height, fps, duration, wordTimings, audioPath }) {
   204→   204→  const ffmpeg = await ensureFfmpeg();
   205→   205→  const browser = await launchBrowser(width, height);
   206→   206→  const tmpDir = fs.mkdtempSync(path.join(os.tmpdir(), "codevid-"));
   207→   207→  const outputPath = path.join(tmpDir, "output.mp4");
   208→   208→  try {
   209→   209→    const page = await browser.newPage();
   210→   210→    page.setDefaultNavigationTimeout(60_000);
   211→   211→    await page.setContent(injectFonts(html), { waitUntil: "load" });
   212→   212→
   213→   213→    const hasRender = await page.evaluate(() => typeof window.renderAtTime === "function" || typeof window.__renderAtTime === "function");
   214→   214→    if (!hasRender) {
   215→   215→      throw new Error('صفحة HTML لا تعرّف window.renderAtTime(t, duration, wordTimings) — أضفها في <script> داخل الصفحة.');
   216→   216→    }
   217→   217→    if (wordTimings) {
   218→   218→      await page.evaluate((wt) => { window.__wordTimings = wt; }, wordTimings);
   219→   219→    }
   220→   220→    // نداء تسخين بالمدة الحقيقية كي لا يرمي renderAtTime خطأ عند التقسيم على المدة
   221→   221→    await page.evaluate((d) => {
   222→   222→      try {
   223→   223→        window.renderAtTime && window.renderAtTime(0, d, window.__wordTimings);
   224→   224→      } catch (e) {
   225→   225→        window.__mekoRenderError = String((e && e.message) || e);
   226→   226→      }
   227→   227→    }, duration);
   228→   228→    const warmError = await page.evaluate(() => window.__mekoRenderError || null);
   229→   229→    if (warmError) throw new Error(`renderAtTime ألقت خطأً عند t=0: ${warmError.slice(0, 300)}`);
   230→   230→
   231→   231→    // ملء الكادر: اكتشاف مقاس المشهد الطبيعي وتكبيره ليملأ 100% من الإطار
   232→   232→    // (المشاهد المصممة 720×1280 أو 540×960 تظهر صغيرة بزاوية الإطار دونه)
   233→   233→    const fit = await fitStageToFrame(page, width, height);
   234→   234→    if (fit) {
   235→   235→      log(`ملء الكادر: المشهد ${fit.naturalW}×${fit.naturalH} → ${width}×${height} عبر ${fit.strategy} (scale=${fit.scale.toFixed(3)})`);
   236→   236→      await writeSummary(`**ملء الكادر**: المشهد ${fit.naturalW}×${fit.naturalH} → ${width}×${height} — ${fit.strategy === "viewport-dsf" ? "تكبير بدقة أعلى (deviceScaleFactor)" : fit.strategy === "transform-cover" ? "تحويل CSS مع اقتصاص طفيف للمواءمة" : "تحويل CSS مع حشو بخلفية المشهد"} (scale ${fit.scale.toFixed(3)})`);
   237→   237→    }
   238→   238→
   239→   239→    // أمر ffmpeg — الإطارات تُضخ عبر stdin أثناء الالتقاط (بلا ملفات وسيطة)
   240→   240→    const args = ["-y", "-hide_banner", "-loglevel", "error", "-progress", "pipe:1", "-nostats"];
   241→   241→    args.push("-f", "image2pipe", "-framerate", String(fps), "-i", "-");
   242→   242→    if (audioPath) {
   243→   243→      args.push("-i", audioPath);
   244→   244→    } else {
   245→   245→      args.push("-f", "lavfi", "-t", duration.toFixed(3), "-i", "anullsrc=channel_layout=stereo:sample_rate=44100");
   246→   246→    }
   247→   247→    args.push("-map", "0:v:0", "-map", "1:a:0");
   248→   248→    // veryfast: ذاكرة أقل بكثير من medium بنفس الجودة البصرية تقريبًا عند CRF 18
   249→   249→    // (قابل للضبط بـ FFMPEG_X264_PRESET للبيئات الغنية بالذاكرة)
   250→   250→    args.push("-c:v", "libx264", "-crf", "18", "-preset", process.env.FFMPEG_X264_PRESET || "veryfast", "-pix_fmt", "yuv420p", "-r", String(fps));
   251→   251→    args.push("-c:a", "aac", "-b:a", "192k", "-ar", "44100");
   252→   252→    args.push("-t", duration.toFixed(3), "-movflags", "+faststart");
   253→   253→    args.push(outputPath);
   254→   254→
   255→   255→    const child = spawn(ffmpeg, args, { stdio: ["pipe", "ignore", "pipe"] });
   256→   256→    let ffmpegErr = "";
   257→   257→    child.stderr.on("data", (d) => {
   258→   258→      ffmpegErr += d.toString();
   259→   259→      if (ffmpegErr.length > 20000) ffmpegErr = ffmpegErr.slice(-20000);
   260→   260→    });
   261→   261→    const closed = new Promise((resolve, reject) => {
   262→   262→      child.on("error", reject);
   263→   263→      child.on("close", (code) => (code === 0 ? resolve() : reject(new Error(`ffmpeg خرج بكود ${code}: ${ffmpegErr.slice(-600)}`))));
   264→   264→    });
   265→   265→
   266→   266→    const frames = Math.round(fps * duration); // معدل ثابت حقيقي — بلا أي تخفيض
   267→   267→    let captured = 0;
   268→   268→    let lastPing = -1;
   269→   269→    for (let i = 0; i < frames; i++) {
   270→   270→      const t = (i * duration) / frames; // == i / fps بالضبط
   271→   271→      await page.evaluate((tt, d) => window.renderAtTime(tt, d, window.__wordTimings), t, duration);
   272→   272→      // PNG بلا فقد — جودة كاملة على عامل قوي بلا قيود زمن
   273→   273→      const buf = await page.screenshot({ type: "png", captureBeyondViewport: false });
   274→   274→      if (!child.stdin.write(buf)) {
   275→   275→        await new Promise((resolve) => child.stdin.once("drain", resolve));
   276→   276→      }
   277→   277→      captured++;
   278→   278→      // تقدم حي على المهمة (10..95) + ملخص الـ workflow كل ~10%
   279→   279→      const pct = Math.round(10 + 85 * ((i + 1) / frames));
   280→   280→      if (pct >= lastPing + 10 || i === frames - 1) {
   281→   281→        lastPing = pct;
   282→   282→        void pingProgress(Math.min(95, pct)).catch(() => {});
   283→   283→        log(`التقدم: ${captured}/${frames} إطارًا (${pct}%)`);
   284→   284→      }
   285→   285→    }
   286→   286→    child.stdin.end();
   287→   287→    await closed;
   288→   288→
   289→   289→    const st = fs.statSync(outputPath);
   290→   290→    if (st.size < 20_000) throw new Error(`الناتج صغير بشكل مريب (${st.size} بايت) — رندر فاشل غالبًا`);
   291→   291→
   292→   292→    // التحقق النهائي بـ ffprobe: الأبعاد والمعدل والمدة ووجود مسار الصوت فعليًا
   293→   293→    const probe = await probeOutput(outputPath, { width, height, fps, duration, expectAudio: Boolean(audioPath) });
   294→   294→    await writeSummary(
   295→   295→      `**تحقق ffprobe**: ${probe.w}×${probe.h} @ ${probe.fps}fps — ${probe.durationSec.toFixed(2)}s — ` +
   296→   296→      (probe.hasAudio
   297→   297→        ? `مسار صوت ${probe.audioCodec}${audioPath ? " (التعليق مدمج)" : " (صامت مولّد — بلا تعليق صوتي مطلوب)"}`
   298→   298→        : "بلا مسار صوت"),
   299→   299→    );
   300→   300→
   301→   301→    fs.copyFileSync(outputPath, PROGRESS_VIDEO_PATH);
   302→   302→    return { frames: captured, size: st.size, probe };
   303→   303→  } finally {
   304→   304→    try { await browser.close(); } catch { /* تجاهل */ }
   305→   305→    fs.promises.rm(tmpDir, { recursive: true, force: true }).catch(() => {});
   306→   306→  }
   307→   307→}

   310→   310→// ملء الكادر — اكتشاف مقاس المشهد الطبيعي وملاءمته للإطار الهدف
   311→   311→
   312→   312→/**
   313→   313→ * المشاهد تُصمّم غالبًا بمقاس أصغر من FHD (720×1280 أو 540×960…) فتظهر صغيرة
   314→   314→ * بزاوية الإطار مع فراغ أسود. هنا نقيس المسرح الفعلي داخل الصفحة ثم نملأ الإطار
   315→   315→ * بأحد استراتيجيتين:
   316→   316→ *  1) viewport-dsf (الأفضل جودةً): إعادة ضبط العرض إلى مقاس التصميم مع
   317→   317→ *     deviceScaleFactor = الهدف/التصميم — المتصفح يرسم النص والمتجهات بدقة
   318→   318→ *     أعلى (ليس تكبيرًا نقطيًا). يُعتمد بعد تحقق فعلي من أبعاد لقطة PNG؛
   319→   319→ *  2) transform: تكبير CSS موحد للمسرح مع توسيط — اقتصاص مركزي عند تفاوت
   320→   320→ *     الأبعاد حتى 15%، أو حشو بخلفية المشهد عند تفاوت أكبر كي لا تظهر أشرطة سوداء.
   321→   321→ */
   322→   322→async function fitStageToFrame(page, targetW, targetH) {
   323→   323→  try {
   324→   324→    const natural = await page.evaluate(detectStageInPage);
   325→   325→    if (!natural || !(natural.w >= 160) || !(natural.h >= 160)) return null;
   326→   326→    // المشهد سائل بحجم الإطار بالفعل — لا شيء لفعله
   327→   327→    if (Math.abs(natural.w - targetW) <= 2 && Math.abs(natural.h - targetH) <= 2) return null;
   328→   328→
   329→   329→    // المحاولة الأولى: deviceScaleFactor — أعلى جودة نص ومتجهات
   330→   330→    const nwR = Math.round(natural.w);
   331→   331→    const nhR = Math.round(natural.h);
   332→   332→    const dsf = targetW / nwR;
   333→   333→    if (dsf > 0.05 && dsf < 8) {
   334→   334→      await page.setViewport({ width: nwR, height: nhR, deviceScaleFactor: dsf });
   335→   335→      await page.evaluate(() => window.dispatchEvent(new Event("resize"))).catch(() => {});
   336→   336→      const probe = await page.screenshot({ type: "png", captureBeyondViewport: false });
   337→   337→      // puppeteer 23 يعيد Uint8Array بدل Buffer — DataView آمنة للاثنين
   338→   338→      const dv = new DataView(probe.buffer, probe.byteOffset, probe.byteLength);
   339→   339→      const pw = dv.getUint32(16); // IHDR width
   340→   340→      const ph = dv.getUint32(20); // IHDR height
   341→   341→      if (pw === targetW && ph === targetH) {
   342→   342→        return { naturalW: nwR, naturalH: nhR, strategy: "viewport-dsf", scale: dsf };
   343→   343→      }
   344→   344→      log(`تحقق dsf فشل (اللقطة ${pw}×${ph} بدل ${targetW}×${targetH}) — التحول إلى تكبير CSS`);
   345→   345→    }
   346→   346→
   347→   347→    // الاستراتيجية الثانية: تكبير CSS موحد مع توسيط (اقتصاص ≤15% أو حشو)
   348→   348→    const applied = await page.evaluate(applyStageTransformInPage, { targetW, targetH });
   349→   349→    if (!applied || !(applied.scale > 0)) {
   350→   350→      await page.setViewport({ width: targetW, height: targetH, deviceScaleFactor: 1 });
   351→   351→      return null;
   352→   352→    }
   353→   353→    return {
   354→   354→      naturalW: Math.round(applied.naturalW),
   355→   355→      naturalH: Math.round(applied.naturalH),
   356→   356→      strategy: applied.mode === "cover" ? "transform-cover" : "transform-contain",
   357→   357→      scale: applied.scale,
   358→   358→    };
   359→   359→  } catch (e) {
   360→   360→    log(`ملء الكادر تخطى: ${e instanceof Error ? e.message.slice(0, 200) : String(e)}`);
   361→   361→    try { await page.setViewport({ width: targetW, height: targetH, deviceScaleFactor: 1 }); } catch { /* تجاهل */ }
   362→   362→    return null;
   363→   363→  }
   364→   364→}

   366→   366→/** كشف المقاس الطبيعي للمشهد داخل الصفحة — يعمل داخل المتصفح ويوسم العناصر المرشحة */
   367→   367→function detectStageInPage() {
   368→   368→  const vw = window.innerWidth;
   369→   369→  const vh = window.innerHeight;
   370→   370→  const vArea = vw * vh;
   371→   371→  const nearly = (a, b) => Math.abs(a - b) <= 2;
   372→   372→
   373→   373→  const marked = [];
   374→   374→  const consider = (el, w, h, priority) => {
   375→   375→    if (!el || !(w >= 160) || !(h >= 160)) return;
   376→   376→    if (nearly(w, vw) && nearly(h, vh)) return; // بحجم الإطار الحالي — مشهد سائل
   377→   377→    // عناصر صغيرة (شارة/شريط ترجمة) لا تمثل المسرح الكامل
   378→   378→    if (w * h < vArea * 0.12 || Math.min(w / vw, h / vh) < 0.3) return;
   379→   379→    marked.push({ el, w, h, priority, area: w * h });
   380→   380→  };
   381→   381→
   382→   382→  // 1) عناصر الوسائط بأبعاد تصميمية صريحة — أعلى أولوية
   383→   383→  for (const el of document.querySelectorAll("svg, canvas, video")) {
   384→   384→    const r = el.getBoundingClientRect();
   385→   385→    consider(el, r.width || el.clientWidth, r.height || el.clientHeight, 1);
   386→   386→  }
   387→   387→  // 2) جسم/جذر بحجم تصميم ثابت (أقل من الإطار في البعدين)
   388→   388→  const body = document.body;
   389→   389→  const root = document.documentElement;
   390→   390→  if (body && body.offsetWidth < vw - 2 && body.offsetHeight < vh - 2) {
   391→   391→    consider(body, body.offsetWidth, body.offsetHeight, 2);
   392→   392→  }
   393→   393→  if (root && root.clientWidth < vw - 2 && root.clientHeight < vh - 2) {
   394→   394→    consider(root, root.clientWidth, root.clientHeight, 2);
   395→   395→  }
   396→   396→  // 3) حاويات مسرح شائعة
   397→   397→  const stageSel = "#stage,[data-stage],.stage,#scene,.scene,#app,#root,#container,#wrapper,.wrapper,main";
   398→   398→  for (const el of document.querySelectorAll(stageSel)) {
   399→   399→    if (el === body || el === root) continue;
   400→   400→    const r = el.getBoundingClientRect();
   401→   401→    consider(el, r.width, r.height, 3);
   402→   402→  }
   403→   403→  // 4) أبناء الجسم المباشرون — الابن الكبير الوحيد هو المسرح غالبًا، وإلا اتحادهم
   404→   404→  const kids = body
   405→   405→    ? Array.from(body.children).filter((c) => !["SCRIPT", "STYLE", "LINK", "NOSCRIPT"].includes(c.tagName))
   406→   406→    : [];
   407→   407→  const bigRects = kids.map((c) => c.getBoundingClientRect()).filter((r) => r.width >= 160 && r.height >= 160);
   408→   408→  if (bigRects.length === 1) {
   409→   409→    const theKid = kids.find((c) => { const r = c.getBoundingClientRect(); return r.width >= 160 && r.height >= 160; });
   410→   410→    consider(theKid, bigRects[0].width, bigRects[0].height, 4);
   411→   411→  } else if (bigRects.length > 1) {
   412→   412→    const u = bigRects.reduce((a, r) => ({
   413→   413→      left: Math.min(a.left, r.left), top: Math.min(a.top, r.top),
   414→   414→      right: Math.max(a.right, r.right), bottom: Math.max(a.bottom, r.bottom),
   415→   415→    }), { left: Infinity, top: Infinity, right: -Infinity, bottom: -Infinity });
   416→   416→    if (u.right > u.left && u.bottom > u.top) {
   417→   417→      // اتحاد عدة أبناء: يوسم كل عنصر كبير كي يُحوّل مجتمعين لاحقًا
   418→   418→      kids.forEach((c) => { const r = c.getBoundingClientRect(); if (r.width >= 160 && r.height >= 160) consider(c, r.width, r.height, 5); });
   419→   419→    }
   420→   420→  }
   421→   421→
   422→   422→  if (!marked.length) return null;
   423→   423→  marked.sort((a, b) => (a.priority - b.priority) || (b.area - a.area));
   424→   424→  const best = marked[0];
   425→   425→  // وسوم على كل المرشحين — لا يمكن إعادة عناصر DOM من evaluate فتعاد قراءتها بالوسم
   426→   426→  for (const m of marked) m.el.setAttribute("data-meko-stage", String(m.priority));
   427→   427→  const r = best.el.getBoundingClientRect();
   428→   428→  return { w: best.w, h: best.h, left: r.left, top: r.top };
   429→   429→}

   431→   431→/** تطبيق التكبير CSS الموحد على العناصر الموسومة — يعمل داخل المتصفح */
   432→   432→function applyStageTransformInPage(spec) {
   433→   433→  const els = Array.from(document.querySelectorAll("[data-meko-stage]"));
   434→   434→  if (!els.length) return null;
   435→   435→  const targetW = spec.targetW;
   436→   436→  const targetH = spec.targetH;
   437→   437→  const rects = els.map((el) => ({ el, r: el.getBoundingClientRect() }));
   438→   438→
   439→   439→  // اتحاد المسرح الموسوم — المرجع للتوسيط والاقتصاص المركزي
   440→   440→  const u = rects.reduce((a, x) => ({
   441→   441→    left: Math.min(a.left, x.r.left), top: Math.min(a.top, x.r.top),
   442→   442→    right: Math.max(a.right, x.r.right), bottom: Math.max(a.bottom, x.r.bottom),
   443→   443→  }), { left: Infinity, top: Infinity, right: -Infinity, bottom: -Infinity });
   444→   444→  const nw = u.right - u.left;
   445→   445→  const nh = u.bottom - u.top;
   446→   446→  if (!(nw >= 160) || !(nh >= 160)) return null;
   447→   447→
   448→   448→  const sx = targetW / nw;
   449→   449→  const sy = targetH / nh;
   450→   450→  const overflowRatio = 1 - Math.min(sx, sy) / Math.max(sx, sy);
   451→   451→  // توافق الأبعاد أو اقتصاص ≤15% → تغطية تامة؛ تفاوت كبير → حشو (حفاظ على المحتوى)
   452→   452→  const mode = overflowRatio <= 0.15 ? "cover" : "contain";
   453→   453→  const s = mode === "cover" ? Math.max(sx, sy) : Math.min(sx, sy);
   454→   454→  const uLeft = (targetW - nw * s) / 2;
   455→   455→  const uTop = (targetH - nh * s) / 2;
   456→   456→
   457→   457→  for (const { el, r } of rects) {
   458→   458→    // ترجمة + تكبير بمنشأ الزاوية العلوية اليسرى — يحفظ التخطيط النسبي لعناصر متعددة
   459→   459→    const dx = uLeft + s * (r.left - u.left) - r.left;
   460→   460→    const dy = uTop + s * (r.top - u.top) - r.top;
   461→   461→    el.style.transformOrigin = "0 0";
   462→   462→    el.style.transform = `translate(${dx.toFixed(2)}px, ${dy.toFixed(2)}px) scale(${s.toFixed(5)})`;
   463→   463→    el.removeAttribute("data-meko-stage");
   464→   464→  }
   465→   465→
   466→   466→  // عند الحشو: خلفية الجذر من خلفية المسرح كي لا تبدو الأشرطة سوداء
   467→   467→  if (mode === "contain") {
   468→   468→    const bg = getComputedStyle(rects[0].el).backgroundColor;
   469→   469→    if (bg && bg !== "rgba(0, 0, 0, 0)") {
   470→   470→      document.documentElement.style.background = bg;
   471→   471→      if (document.body) document.body.style.background = bg;
   472→   472→    }
   473→   473→  }
   474→   474→  return { naturalW: nw, naturalH: nh, scale: s, mode, count: rects.length };
   475→   475→}

   502→   502→/** كروم النظام على ubuntu-latest (google-chrome-stable مثبت مسبقًا)، وإن غاب يُثبت Chrome for Testing */
   503→   503→async function resolveChrome() {
   504→   504→  const candidates = ["/usr/bin/google-chrome-stable", "/usr/bin/google-chrome", "/usr/bin/chromium-browser", "/usr/bin/chromium"];
   505→   505→  for (const c of candidates) {
   506→   506→    if (fs.existsSync(c)) return c;
   507→   507→  }
   508→   508→  try {
   509→   509→    const found = execFileSync("which", ["google-chrome-stable"], { encoding: "utf8" }).trim();
   510→   510→    if (found) return found;
   511→   511→  } catch { /* نكمل للتثبيت */ }
   512→   512→  log("كروم النظام غير موجود — تثبيت Chrome for Testing…");
   513→   513→  execFileSync("npx", ["-y", "@puppeteer/browsers", "install", "chrome@stable", "--path", "/tmp/chrome"], { stdio: "inherit", timeout: 240_000 });
   514→   514→  const root = "/tmp/chrome";
   515→   515→  const stack = [root];
   516→   516→  while (stack.length) {
   517→   517→    const dir = stack.pop();
   518→   518→    for (const name of fs.readdirSync(dir)) {
   519→   519→      const p = path.join(dir, name);
   520→   520→      const st = fs.statSync(p);
   521→   521→      if (st.isDirectory()) stack.push(p);
   522→   522→      else if (name === "chrome") return p;
   523→   523→    }
   524→   524→  }
   525→   525→  throw new Error("تعذر العثور على كروم بعد التثبيت");
   526→   526→}

   528→   528→/** حقن خطي Meko (Noto Sans Arabic + Liberation Sans) كـ @font-face — نفس سلوك محرك Vercel */
   529→   529→function injectFonts(html) {
   530→   530→  const fontsDir = path.join(__dirname, "..", "merger", "fonts");
   531→   531→  const families = { MekoArabic: "NotoSansArabic-Regular.ttf", MekoLatin: "LiberationSans-Regular.ttf" };
   532→   532→  const faces = [];
   533→   533→  const chain = [];
   534→   534→  for (const [family, file] of Object.entries(families)) {
   535→   535→    try {
   536→   536→      const bytes = fs.readFileSync(path.join(fontsDir, file));
   537→   537→      if (bytes.length > 1000) {
   538→   538→        faces.push(`@font-face{font-family:'${family}';src:url('data:font/ttf;base64,${bytes.toString("base64")}') format('truetype');font-weight:normal;font-style:normal;font-display:block;}`);
   539→   539→        chain.push(`'${family}'`);
   540→   540→      }
   541→   541→    } catch { /* الخط غائب — نكمل بخطوط النظام */ }
   542→   542→  }
   543→   543→  const fallback = chain.length ? `html,body{font-family:${chain.join(",")},system-ui,-apple-system,'Segoe UI',sans-serif;}` : "";
   544→   544→  if (!faces.length && !fallback) return html;
   545→   545→  const tag = `<style data-meko-fonts>${faces.join("")}${fallback}</style>`;
   546→   546→  if (/<\/head>/i.test(html)) return html.replace(/<\/head>/i, `${tag}</head>`);
   547→   547→  if (/<body[^>]*>/i.test(html)) return html.replace(/<body[^>]*>/i, (m) => `${m}${tag}`);
   548→   548→  return tag + html;
   549→   549→}

   568→   568→async function audioDurationSec(filePath) {
   569→   569→  return new Promise((resolve) => {
   570→   570→    const child = spawn("ffmpeg", ["-hide_banner", "-i", filePath, "-f", "null", "-"], { stdio: ["ignore", "ignore", "pipe"] });
   571→   571→    let stderr = "";
   572→   572→    child.stderr.on("data", (d) => (stderr += d.toString()));
   573→   573→    child.on("error", () => resolve(0));
   574→   574→    child.on("close", () => {
   575→   575→      const m = stderr.match(/Duration:\s*(\d+):(\d{2}):(\d{2})\.(\d{2})/);
   576→   576→      if (m) resolve(Number(m[1]) * 3600 + Number(m[2]) * 60 + Number(m[3]) + Number(m[4]) / 100);
   577→   577→      else resolve(0);
   578→   578→    });
   579→   579→  });
   580→   580→}

   598→   598→/** بديل بلا ffprobe — تحليل تقرير ffmpeg عن المسارات والمدة */
   599→   599→function ffmpegInfoFallback(file) {
   600→   600→  const r = spawnSync("ffmpeg", ["-hide_banner", "-i", file, "-f", "null", "-"], { encoding: "utf8", timeout: 60_000 });
   601→   601→  const err = String((r && r.stderr) || "");

   603→   603→  const durationSec = dur ? Number(dur[1]) * 3600 + Number(dur[2]) * 60 + Number(dur[3]) + Number(dur[4]) / 100 : 0;
   604→   604→  const vid = /Stream #\d+:\d+.*?: Video: .*?(\d{2,5})x(\d{2,5})/.exec(err);

   607→   607→  return {
   608→   608→    video: vid ? { width: Number(vid[1]), height: Number(vid[2]), fpsText: fpsM ? fpsM[1] : "" } : null,
   609→   609→    audio: aud ? { codec_name: aud[1] } : null,
   610→   610→    durationSec,
   611→   611→  };

   614→   614→function parseFpsText(text) {
   615→   615→  if (!text) return 0;
   616→   616→  const m = String(text).match(/^(\d+)(?:\/(\d+))?$/);
   617→   617→  if (m) return Number(m[2]) ? Number(m[1]) / Number(m[2]) : Number(m[1]);
   618→   618→  const n = Number(text);
   619→   619→  return Number.isFinite(n) ? n : 0;
   620→   620→}

   623→   623→ * تحقق صارم من الناتج النهائي قبل النشر — أي انحراف يعني رندرًا فاشلًا:
   624→   624→ * أبعاد الإطار = المطلوب، المعدل = المطلوب، المدة قريبة، ومسار الصوت موجود

   627→   627→async function probeOutput(outputPath, expect) {
   628→   628→  const j = ffprobeJson(outputPath);
   629→   629→  let video = null;
   630→   630→  let audio = null;
   631→   631→  let durationSec = 0;
   632→   632→  if (j && Array.isArray(j.streams)) {
   633→   633→    video = j.streams.find((s) => s.codec_type === "video") || null;
   634→   634→    audio = j.streams.find((s) => s.codec_type === "audio") || null;
   635→   635→    durationSec = Number((j.format && j.format.duration) || 0) || Number((video && video.duration) || 0) || 0;
   636→   636→  } else {
   637→   637→    const fb = ffmpegInfoFallback(outputPath);
   638→   638→    video = fb.video;
   639→   639→    audio = fb.audio;
   640→   640→    durationSec = fb.durationSec;
   641→   641→  }
   642→   642→  if (!video) throw new Error("تحقق الناتج: لا يوجد مسار فيديو في MP4 النهائي!");
   643→   643→
   644→   644→  const w = Number(video.width) || 0;
   645→   645→  const h = Number(video.height) || 0;
   646→   646→  const fps = parseFpsText(video.avg_frame_rate || video.r_frame_rate || video.fpsText || "");
   647→   647→  if (w !== expect.width || h !== expect.height) {
   648→   648→    throw new Error(`تحقق الناتج: الأبعاد ${w}×${h} بدل ${expect.width}×${expect.height} المطلوبة`);
   649→   649→  }
   650→   650→  if (fps && Math.abs(fps - expect.fps) > 0.6) {
   651→   651→    throw new Error(`تحقق الناتج: معدل الإطارات ${fps.toFixed(2)} بدل ${expect.fps} المطلوب`);
   652→   652→  }
   653→   653→  const durTol = Math.max(0.6, expect.duration * 0.05);
   654→   654→  if (durationSec && Math.abs(durationSec - expect.duration) > durTol) {
   655→   655→    throw new Error(`تحقق الناتج: المدة ${durationSec.toFixed(2)}s بدل ${expect.duration.toFixed(2)}s المطلوبة`);
   656→   656→  }
   657→   657→  if (expect.expectAudio && !audio) {
   658→   658→    throw new Error("تحقق الناتج: الصوت المدمج اختفى من MP4 النهائي — لا يوجد مسار صوت!");
   659→   659→  }
   660→   660→  return {
   661→   661→    w,
   662→   662→    h,
   663→   663→    fps: fps || expect.fps,
   664→   664→    durationSec: durationSec || expect.duration,
   665→   665→    hasAudio: Boolean(audio),
   666→   666→    audioCodec: (audio && audio.codec_name) || null,
   667→   667→  };
   668→   668→}

   670→   670→async function downloadAudio(url) {
   671→   671→  const str = String(url);
   672→   672→
   673→   673→  // 1) صوت مضمّن data: (الحل الأخير للرفع في Meko عندما تفشل كل المضيفين) —
   674→   674→  //    يُفك محليًا بلا أي شبكة، فيصل الصوت دائمًا ولو حُجب كل المضيفين الخارجيين
   675→   675→  if (/^data:/i.test(str)) {
   676→   676→    const m = str.match(/^data:([\w.+/-]+)?;base64,([\s\S]+)$/i);
   677→   677→    if (!m) throw new Error("رابط الصوت المضمّن (data:) غير صالح — صيغة base64 مطلوبة");
   678→   678→    const buf = Buffer.from(m[2].replace(/\s+/g, ""), "base64");
   679→   679→    if (buf.length < 512) throw new Error(`الصوت المضمّن (data:) أصغر من اللازم (${buf.length} بايت) — قد يكون فارغًا`);
   680→   680→    const dest = path.join(os.tmpdir(), `audio-${JOB_ID}.bin`);
   681→   681→    fs.writeFileSync(dest, buf);
   682→   682→    log(`الصوت المضمّن (data:${m[1] || "audio"}) — ${(buf.length / 1024).toFixed(0)}KB`);
   683→   683→    return dest;
   684→   684→  }
   685→   685→
   686→   686→  // 2) رابط http — محاولتان مع ترويسة متصفح (مضيفون مثل catbox يرفضون ترويسة Node)
   687→   687→  const UA = "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/126.0.0.0 Safari/537.36";
   688→   688→  let lastErr = null;
   689→   689→  for (let attempt = 1; attempt <= 2; attempt++) {
   690→   690→    try {
   691→   691→      const res = await fetchWithTimeout(str, { headers: { "User-Agent": UA, Accept: "*/*" } }, 120_000);
   692→   692→      if (!res.ok) throw new Error(`تعذر تنزيل الصوت (HTTP ${res.status}) من ${str.slice(0, 120)}`);
   693→   693→      const buf = Buffer.from(await res.arrayBuffer());
   694→   694→      if (buf.length < 512) throw new Error("ملف الصوت غير صالح أو فارغ");
   695→   695→      const head = buf.subarray(0, 512).toString("latin1").trim().toLowerCase();
   696→   696→      const ct = (res.headers.get("content-type") || "").toLowerCase();
   697→   697→      if (head.startsWith("<!doctype") || head.startsWith("<html") || ct.includes("text/html")) {
   698→   698→        throw new Error(`رابط الصوت يعيد صفحة HTML بدل الملف الخام (${ct || "بلا نوع"}) — انتهت صلاحية الرابط: ${str.slice(0, 120)}`);
   699→   699→      }
   700→   700→      const dest = path.join(os.tmpdir(), `audio-${JOB_ID}.bin`);
   701→   701→      fs.writeFileSync(dest, buf);
   702→   702→      return dest;
   703→   703→    } catch (e) {
   704→   704→      lastErr = e;
   705→   705→      if (attempt < 2) {
   706→   706→        log(`تنزيل الصوت فشل (محاولة ${attempt}): ${e instanceof Error ? e.message.slice(0, 160) : String(e).slice(0, 160)} — إعادة بعد 3 ثوانٍ…`);
   707→   707→        await new Promise((r) => setTimeout(r, 3000));
   708→   708→      }
   709→   709→    }
   710→   710→  }
   711→   711→  throw lastErr instanceof Error ? lastErr : new Error(String(lastErr));
   712→   712→}

   717→   717→async function publishToMerger(meta) {
   718→   718→  try {
   719→   719→    const buf = fs.readFileSync(PROGRESS_VIDEO_PATH);
   720→   720→    const chunks = Math.ceil(buf.length / CHUNK_BYTES);
   721→   721→    for (let i = 0; i < chunks; i++) {
   722→   722→      const part = buf.subarray(i * CHUNK_BYTES, Math.min((i + 1) * CHUNK_BYTES, buf.length));
   723→   723→      const res = await fetchWithTimeout(
   724→   724→        `${MERGER_BASE}/api/render-complete?op=chunk&task_id=${TASK_ID}&job_id=${JOB_ID}&chunk=${i}`,
   725→   725→        { method: "POST", headers: { "Content-Type": "application/octet-stream" }, body: part },
   726→   726→        60_000,
   727→   727→      );
   728→   728→      if (!res.ok) throw new Error(`فشل رفع الجزء ${i + 1}/${chunks} (HTTP ${res.status}): ${(await res.text()).slice(0, 200)}`);
   729→   729→      log(`رفع الجزء ${i + 1}/${chunks} — ${(part.length / 1048576).toFixed(2)}MB`);
   730→   730→    }
   731→   731→    const done = await finalizeToMerger(meta);
   732→   732→    if (!done) return null;

   734→   734→    const jobRes = await fetchWithTimeout(`${MERGER_BASE}/api/jobs/${JOB_ID}`, {}, 30_000);
   735→   735→    const videoUrl = done.videoUrl.replace(/\?download=1$/, "");
   736→   736→    const videoRes = await fetchWithTimeout(videoUrl, { headers: { Range: "bytes=0-31" } }, 60_000);
   737→   737→    const contentType = videoRes.headers.get("content-type") || "";
   738→   738→    const videoOk = (videoRes.status === 200 || videoRes.status === 206) && contentType.includes("video");
   739→   739→    await videoRes.arrayBuffer().catch(() => undefined);
   740→   740→    if (!videoOk) throw new Error(`تحقق الفيديو فشل (HTTP ${videoRes.status}, content-type: ${contentType})`);
   741→   741→    log(`تحقق النشر: صف المهمة HTTP ${jobRes.status} — الفيديو HTTP ${videoRes.status} مع content-type video`);
   742→   742→    return done;
   743→   743→  } catch (e) {
   744→   744→    log(`فشل النشر عبر meko-manger: ${e instanceof Error ? e.message : String(e)}`);
   745→   745→    return null;
   746→   746→  }
   747→   747→}

   889→   889→function sanitizeTimings(raw) {
   890→   890→  if (!Array.isArray(raw) || raw.length < 2) return null;
   891→   891→  const list = raw.slice(0, MAX_WORD_TIMINGS).map((w) => {
   892→   892→    const o = w && typeof w === "object" ? w : {};
   893→   893→    const start = Number(o.start);
   894→   894→    const end = Number(o.end);
   895→   895→    const text = typeof o.text === "string" ? o.text.slice(0, 200) : "";
   896→   896→    if (!Number.isFinite(start) || !Number.isFinite(end) || end < start || !text) return null;
   897→   897→    return { start: Math.max(0, start), end: Math.max(0, end), text };
   898→   898→  }).filter(Boolean);
   899→   899→  if (list.length < 2) return null;
   900→   900→  list.sort((a, b) => a.start - b.start);
   901→   901→  return list;
   902→   902→}
