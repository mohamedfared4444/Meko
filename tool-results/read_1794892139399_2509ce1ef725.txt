    77→/** إطلاق workflow الرندر الكامل — 204 = نجح الإطلاق، غير ذلك فشل حتمي معلن */
    78→async function dispatchRenderWorkflow(token: string, inputs: Record<string, string>): Promise<void> {
    79→  const url = `https://api.github.com/repos/${GH_RENDER_REPO}/actions/workflows/${GH_WORKFLOW_FILE}/dispatches`;
    80→  const res = await fetch(url, {
    81→    method: "POST",
    82→    headers: {
    83→      Authorization: `Bearer ${token}`,
    84→      Accept: "application/vnd.github+json",
    85→      "X-GitHub-Api-Version": "2022-11-28",
    86→      "Content-Type": "application/json",
    87→    },
    88→    body: JSON.stringify({ ref: GH_RENDER_REF, inputs }),
    89→  });
    90→  if (res.status === 204) return;
    91→  const text = await res.text().catch(() => "");
    92→  const hint =
    93→    res.status === 401 || res.status === 403
    94→      ? " — تأكد أن GITHUB_TOKEN صالح وبصلاحية repo/workflow وأن ملف الـ workflow موجود على الفرع المطلوب"
    95→      : res.status === 422
    96→        ? " — ملف الـ workflow غير موجود أو مدخلاته لا تطابق (تأكد من رفع .github/workflows/render-code-video.yml على الفرع)"
    97→        : "";
    98→  throw new MergerError(
    99→    `فشل إطلاق الرندر على GitHub Actions (HTTP ${res.status})${hint}: ${text.slice(0, 200)}`,
   100→    res.status,
   101→    true,
   102→  );
   103→}

   116→/** مقاس الفيديو لكل اتجاه:
   117→ *  - mode "fhd" (الرندر الكامل على GitHub Actions): 1080×1920 / 1920×1080 / 1080×1080 بجودة كاملة
   118→ *  - mode "fast" (الرندر الاحتياطي داخل مهلة Vercel): 540×960 / 960×540 / 640×640
   119→ *    — يقلل بكسلات Chromium Software Rasterizer ~45% فيسرّع الالتقاط */
   120→export function dimensionsFor(aspectRatio: "9:16" | "16:9" | "1:1", mode: "fast" | "fhd" = "fast"): { width: number; height: number } {
   121→  if (mode === "fhd") {
   122→    switch (aspectRatio) {
   123→      case "16:9":
   124→        return { width: 1920, height: 1080 };
   125→      case "1:1":
   126→        return { width: 1080, height: 1080 };
   127→      default:
   128→        return { width: 1080, height: 1920 };
   129→    }
   130→  }
   131→  switch (aspectRatio) {
   132→    case "16:9":
   133→      return { width: 960, height: 540 };
   134→    case "1:1":
   135→      return { width: 640, height: 640 };
   136→    default:
   137→      return { width: 540, height: 960 };
   138→  }
   139→}

   209→/** تحقق وتطبيع مدخلات الفيديو البرمجي — رسائل عربية واضحة عند الخطأ */
   210→export function buildCodeVideoParams(input: CodeVideoInput): { ok: true; params: CodeVideoParams } | { ok: false; error: string } {
   211→  const title = (input.title ?? "").trim();
   212→  if (!title) return { ok: false, error: "❌ معطيات غير صالحة: عنوان الفيديو (title) مطلوب." };
   213→  if (title.length > TITLE_MAX) return { ok: false, error: `❌ العنوان طويل جدًا (${title.length} حرفًا) — الحد ${TITLE_MAX} حرفًا.` };
   214→
   215→  const htmlCode = (input.htmlCode ?? "").trim();
   216→  if (htmlCode.length < HTML_CODE_MIN) {
   217→    return { ok: false, error: `❌ html_code قصير جدًا (${htmlCode.length} حرفًا) — مرّر صفحة HTML/CSS/SVG/JS كاملة ذاتية الاحتواء.` };
   218→  }
   219→  if (htmlCode.length > HTML_CODE_MAX) {
   220→    return { ok: false, error: `❌ html_code طويل جدًا (${htmlCode.length} حرفًا) — الحد ${HTML_CODE_MAX} حرفًا.` };
   221→  }
   222→  if (!/renderAtTime/.test(htmlCode)) {
   223→    return {
   224→      ok: false,
   225→      error: "❌ html_code لا يعرّف renderAtTime — أضف <script> يعرّف window.renderAtTime(t, duration, wordTimings) لرسم حالة المشهد في أي لحظة t.",
   226→    };
   227→  }
   228→
   229→  const script = (input.script ?? "").trim();
   230→  if (script.length > SCRIPT_MAX) {
   231→    return { ok: false, error: `❌ script طويل جدًا (${script.length} حرفًا) — الحد ${SCRIPT_MAX} حرفًا (حد التحويل الصوتي نفسه).` };
   232→  }
   233→
   234→  const aspectRatio = input.aspectRatio ?? "9:16";
   235→  if (!["9:16", "16:9", "1:1"].includes(aspectRatio)) {
   236→    return { ok: false, error: "❌ aspect_ratio يقبل 9:16 أو 16:9 أو 1:1 فقط." };
   237→  }
   238→  const durationSeconds = Math.max(DURATION_MIN, Math.min(DURATION_MAX, Math.round(input.durationSeconds || DEFAULT_DURATION)));
   239→  const { width, height } = dimensionsFor(aspectRatio);
   240→  const params: CodeVideoParams = {
   241→    title,
   242→    htmlCode,
   243→    voice: normalizeVoice(input.voice),
   244→    aspectRatio,
   245→    width,
   246→    height,
   247→    fps: DEFAULT_FPS,
   248→    durationSeconds,
   249→    stage: "audio",
   250→    failCount: 0,
   251→  };
   252→  if (script) params.script = script;
   253→  return { ok: true, params };
   254→}

   259→async function ensureAudio(taskId: string, params: CodeVideoParams, deadline: number, onProgress: (p: number, msg: string) => Promise<void> | void): Promise<boolean> {
   260→  if (params.audioUrl && params.audioDuration && params.audioDuration > 0) return true;
   261→  if (!params.script) return true; // لا تعليق صوتي — الرندر الصام مباشرة
   262→  if (Date.now() > deadline - MIN_ACTION_BUDGET_MS) return false; // نافذة ضيقة — أجّل للنداء التالي
   263→
   264→  await onProgress(8, "توليد التعليق الصوتي عبر Edge-TTS مع توقيتات الكلمات…");
   265→  const speech = await synthesizeSpeech(params.script, { voice: params.voice, timeoutMs: TTS_TIMEOUT_MS, collectTimings: true });
   266→  // حراسة: صوت فارغ/شبه فارغ أو بلا مدة يعني فشل التوليد الصامت — يُرفض فورًا
   267→  // بدل أن يمضي للرندر ويخرج فيديو أخرس يبدو مكتملًا
   268→  if (!speech.mp3 || speech.mp3.length < 1000) {
   269→    throw new Error(`الصوت المولّد من Edge-TTS فارغ أو شبه فارغ (${speech.mp3?.length ?? 0} بايت) — أعد المحاولة أو غيّر الصوت`);
   270→  }
   271→  if (!(speech.durationSec > 0.5)) {
   272→    throw new Error(`مدة الصوت المولّد غير صالحة (${speech.durationSec}s) — أعد المحاولة`);
   273→  }
   274→  const url = await uploadViaHosts(speech.mp3, "audio/mpeg", () => Math.max(0, deadline - Date.now()));
   275→  params.audioUrl = url;
   276→  params.audioDuration = Math.round(speech.durationSec * 100) / 100;
   277→  if (speech.words && speech.words.length >= 2) params.wordTimings = speech.words;
   278→  params.stage = "render";
   279→  await saveParams(taskId, params, { progress: 25, error_message: null });
   280→  return true;
   281→}

   286→
   287→async function ensureRender(taskId: string, params: CodeVideoParams, deadline: number, onProgress: (p: number, msg: string) => Promise<void> | void): Promise<boolean> {
   288→  if (params.videoUrl) return true;
   289→
   290→  if (params.renderDispatched) {
   291→    // مُطلق سابقًا على العامل الخارجي — بانتظاره بغض النظر عن توفر التوكن الآن
   292→    // (الرندر الاحتياطي لن يعيد رندل مهمة مُطلقة خارجيًا)، مع حارس ركود يفشلها
   293→    // نهائيًا بعد المهلة
   294→    const at = Date.parse(params.dispatchedAt || "");
   295→    if (Number.isFinite(at) && Date.now() - at > GH_RENDER_STALE_MS) {
   296→      throw new MergerError(
   297→        `تجاوز الرندر الكامل على GitHub Actions مهلة الانتظار (${Math.round(GH_RENDER_STALE_MS / 60000)} دقيقة منذ الإطلاق) — راجع سجل الـ workflow ثم أعد إطلاقه يدويًا بالمعرفين نفسهما إن لزم.`,
   298→        undefined,
   299→        true,
   300→      );
   301→    }
   302→    await saveParams(taskId, params, { progress: 55, error_message: null });
   303→    return false;
   304→  }
   305→
   306→  const token = await getGitHubToken();
   307→  if (token) {
   308→    // الصوت: روابط http تُمرر أيضًا كمدخل إطلاق (احتياط لو فشل جلب الحمولة)؛
   309→    // أما الصوت المضمّن (data: — الحل الأخير للرفع) فيمر حصرًا داخل الحمولة
   310→    // المخزّنة على خدمة الدمج لأن حجمه يجاوز حد مدخلات GitHub (~64KB)
   311→    const audioHttp = params.audioUrl && /^https?:\/\//i.test(params.audioUrl) ? params.audioUrl : null;
   312→    const audioInline = params.audioUrl && !audioHttp ? params.audioUrl : null;
   313→    await onProgress(50, `تجهيز الحمولة وإطلاق الرندر الكامل FHD 30fps على GitHub Actions…${params.audioUrl ? " (مع التعليق الصوتي)" : " (بلا تعليق صوتي)"}`);
   314→    // أبعاد الجودة الكاملة والمعدل الثابت الناعم — يفرضان حتى على المهام القديمة
   315→    // المُنشئة بأبعاد الرندر السريع كي يخرج العامل دائمًا بـ FHD
   316→    const fhd = dimensionsFor(params.aspectRatio, "fhd");
   317→    params.width = fhd.width;
   318→    params.height = fhd.height;
   319→    params.fps = GH_RENDER_FPS;
   320→    const ghJobId = params.ghJobId ?? params.videoJobId ?? randomUUID();
   321→    params.ghJobId = ghJobId;
   322→    params.renderDispatched = true;
   323→    params.dispatchedAt = new Date().toISOString();
   324→
   325→    // كود المشهد يتجاوز حد مدخلات GitHub (~64KB) غالبًا — تُخزّن الحمولة على
   326→    // خدمة الدمج ويُمرر رابطها، وإن تعذر التخزين وكود صغير فيُمرر base64
   327→    const inputs: Record<string, string> = {
   328→      task_id: taskId,
   329→      job_id: ghJobId,
   330→      width: String(params.width),
   331→      height: String(params.height),
   332→      fps: String(GH_RENDER_FPS),
   333→      ...(params.durationSeconds ? { duration: String(params.durationSeconds) } : {}),
   334→      ...(audioHttp ? { audio_url: audioHttp } : {}),
   335→    };
   336→    let payloadStored = false;
   337→    if (mergerConfigured()) {
   338→      try {
   339→        const stored = await mergerStoreRenderPayload({
   340→          task_id: taskId,
   341→          job_id: ghJobId,
   342→          title: params.title,
   343→          html_code: params.htmlCode,
   344→          audio_url: params.audioUrl ?? null,
   345→          word_timings: params.wordTimings ?? null,
   346→          duration_seconds: params.durationSeconds ?? null,
   347→        });
   348→        inputs.payload_url = stored.payloadUrl;
   349→        payloadStored = true;
   350→      } catch {
   351→        /* تُجرب مسار base64 أدناه */
   352→      }
   353→    }
   354→    if (!payloadStored) {
   355→      // الصوت المضمّن (data:) لا يُمرر في مدخلات الإطلاق (يجاوز حد ~64KB ويفشل
   356→      // الإطلاق كله) — بلا تخزين للحمولة يعني إطلاقًا بلا صوت ينتج فيديو أخرس،
   357→      // فنُلغي بدل تسليم ناتج خاطئ، وتُعاد المحاولة عند التحديث القادم
   358→      if (audioInline) {
   359→        throw new MergerError(
   360→          "تعذر تخزين حمولة الرندر (الكود + الصوت المضمّن) على خدمة الدمج — إطلاق بلا الصوت ينتج فيديو أخرس فأُلغي. أعد المحاولة، وإن تكررت فالحجم يتجاوز حد خدمة الدمج.",
   361→          undefined,
   362→          false,
   363→        );
   364→      }
   365→      if (params.htmlCode.length > GH_HTML_B64_LIMIT) {
   366→        throw new MergerError(
   367→          `تعذر تخزين حمولة المشهد على خدمة الدمج وكوده (${params.htmlCode.length} حرفًا) يتجاوز حد مدخلات الإطلاق (${GH_HTML_B64_LIMIT}) — أعد المحاولة لاحقًا.`,
   368→          undefined,
   369→          true,
   370→        );
   371→      }
   372→      inputs.html_code_b64 = Buffer.from(params.htmlCode, "utf8").toString("base64");
   373→    }
   374→
   375→    // احفظ حالة الإطلاق قبل النداء — أي فشل لاحق يصبح مرئيًا على المهمة نفسها
   376→    await saveParams(taskId, params, { progress: 52, error_message: null });
   377→    await dispatchRenderWorkflow(token, inputs);
   378→    return false; // غير متزامن — العامل يكمل المهمة في Turso مباشرة (completed/100/رابط الفيديو)
   379→  }
   380→
   381→  // المسار الاحتياطي (بلا توكن): رندر سريع داخل مهلة Vercel كما كان
   382→  if (Date.now() > deadline - CODEVIDEO_MIN_BUDGET_MS) {
   383→    await saveParams(taskId, params, { progress: 45, error_message: null });
   384→    return false;
   385→  }
   386→  await onProgress(50, "رندر إطارات المشهد عبر Chromium ودمجها بـ ffmpeg (رندر سريع — الرندر الكامل FHD يتطلب GITHUB_TOKEN)…");
   387→  const result = await mergerCodeVideo({
   388→    html_code: params.htmlCode,
   389→    audio_url: params.audioUrl ?? null,
   390→    // المدة: مدة الصوت تحكم عند وجودها (الخدمة تكتشفها تلقائيًا) — وإلا المدة المطلوبة
   391→    ...(params.audioUrl ? {} : { duration_seconds: params.durationSeconds ?? DEFAULT_DURATION }),
   392→    fps: params.fps,
   393→    width: params.width,
   394→    height: params.height,
   395→    // التوقيتات تُرسل فقط عند وجودها — لا null (خدمة الدمج ترفض قيمًا غير مصفوفة)
   396→    ...(params.wordTimings ? { word_timings: params.wordTimings } : {}),
   397→  });
   398→  params.videoUrl = result.videoUrl;
   399→  params.videoDownloadUrl = result.downloadUrl;
   400→  params.videoDuration = result.duration;
   401→  params.videoJobId = result.jobId;
   402→  params.videoFps = result.fps;
   403→  params.videoFrames = result.frames;
   404→  params.renderHost = "vercel";
   405→  await saveParams(taskId, params, {
   406→    progress: 100,
   407→    status: "completed",
   408→    result_urls_json: JSON.stringify([result.videoUrl]),
   409→    error_message: null,
   410→  });
   411→  return true;
   412→}

   417→type PipelineOutcome =
   418→  | { kind: "completed" }

   422→/** تقدّم آلة الحالات خطوة واحدة قدر الإمكان ضمن الميزانية الزمنية المتبقية */
   423→async function advanceCodeVideo(
   424→  taskId: string,
   425→  params: CodeVideoParams,
   426→  budgetMs: number,
   427→  onProgress: (p: number, msg: string) => Promise<void> | void,
   428→): Promise<PipelineOutcome> {
   429→  const deadline = Date.now() + budgetMs;
   430→  try {
   431→    const audioReady = await ensureAudio(taskId, params, deadline, onProgress);
   432→    if (!audioReady) return { kind: "processing", progress: params.videoUrl ? 90 : 20 };
   433→
   434→    const rendered = await ensureRender(taskId, params, deadline, onProgress);
   435→    if (!rendered) {
   436→      // في الانتظار: مُطلق على GitHub Actions (55%) أو مؤجل لنافذة ضيقة (45%)
   437→      return { kind: "processing", progress: params.renderDispatched ? 55 : 45 };
   438→    }
   439→    return { kind: "completed" };
   440→  } catch (e) {
   441→    const msg = e instanceof Error ? e.message : String(e);
   442→    params.failCount = (params.failCount ?? 0) + 1;
   443→    params.lastError = msg.slice(0, 400);
   444→    // خطأ الرندر المعلن من الخدمة نهائي — أخطاء الشبكة المؤقتة تُعاد حتى 3 مرات
   445→    const declaredFinal = e instanceof MergerError && e.declared;
   446→    if (declaredFinal || params.failCount >= 3) {
   447→      await saveParams(taskId, params, { status: "failed", error_message: msg.slice(0, 490) });
   448→      return { kind: "failed", error: msg };
   449→    }
   450→    await saveParams(taskId, params, { status: "processing", error_message: `محاولة متعددة (${params.failCount}): ${msg.slice(0, 300)}` });
   451→    return { kind: "processing", progress: Math.max(5, (params.failCount ?? 0) * 3) };
   452→  }
   453→}

   466→  const params = parseParams(task);
   467→  if (!params) {

   491→    "",
   492→    `task_id: ${task.id}`,
   493→  ];
   494→  return lines.filter((l) => l !== "").join("\n");

   497→/** التشغيل الأول — أداة render_code_video ونقطة /api/tasks/create */
   498→export async function runCodeVideo(opts: CodeVideoInput & {
   499→  waitSeconds: number;
   500→  onProgress?: (progress: number, message: string) => Promise<void> | void;
   501→}): Promise<
   502→  | { kind: "completed"; taskId: string; urls: string[]; text: string }
   503→  | { kind: "processing"; taskId: string; progress: number; text: string }
   504→  | { kind: "failed"; taskId: string; text: string }
   505→  | { kind: "invalid"; taskId?: string; text: string }
   506→> {
   507→  const built = buildCodeVideoParams(opts);
   508→  const rawTitle = (opts.title ?? "").trim();
   509→  const taskId = randomUUID();
   510→
   511→  // المهمة تُسجَّل في Turso فور بدء التشغيل — قبل أي فحص شبكي أو استدعاء خدمة
   512→  // الدمج أو توليد الصوت — حتى تظهر فورًا في شاشة العمل المباشر، ولو فشل
   513→  // الرندر لاحقًا فسبب الفشل مرئي على المهمة نفسها بدل اختفاء الطلب بلا أثر.
   514→  await insertTask({
   515→    id: taskId,
   516→    model: CODE_VIDEO_MODEL,
   517→    type: "video" as MediaType,
   518→    prompt: `فيديو برمجي: ${rawTitle.slice(0, 1900) || "بلا عنوان"}`.slice(0, 2000),
   519→    // فشل التحقق: لا نخزّن الكود الأصلي (قد يكون ضخمًا ويُفشل الإدراج) — الحالة فقط
   520→    paramsJson: JSON.stringify(
   521→      built.ok
   522→        ? built.params
   523→        : { title: rawTitle.slice(0, TITLE_MAX), htmlCode: "", failCount: 1, lastError: built.error.slice(0, 400) } satisfies Partial<CodeVideoParams>,
   524→    ),
   525→    status: "processing",
   526→  });
   527→
   528→  if (!built.ok) {
   529→    const err = built.error.slice(0, 490);
   530→    await updateTask(taskId, { status: "failed", error_message: err });
   531→    return { kind: "invalid", taskId, text: built.error };
   532→  }
   533→  const params = built.params;
   534→
   535→  if (!mergerConfigured()) {
   536→    const err = "❌ خدمة الدمج meko-manger غير مهيأة على هذا الخادم (MERGER_API_BASE غير مضبوط) — الرندر البرمجي يحتاجها لتحويل المشهد إلى فيديو.";
   537→    await updateTask(taskId, { status: "failed", error_message: err.slice(0, 490) });
   538→    return { kind: "invalid", taskId, text: err };
   539→  }
   540→
   541→  const waitSec = Math.max(0, Math.min(opts.waitSeconds, PIPELINE_BUDGET_MS / 1000));
   542→  const outcome = await advanceCodeVideo(taskId, params, Math.min(PIPELINE_BUDGET_MS, waitSec * 1000 || PIPELINE_BUDGET_MS), opts.onProgress ?? (() => {}));
   543→
   544→  if (outcome.kind === "completed") {
   545→    const fresh = (await getTask(taskId))!;
   546→    return { kind: "completed", taskId, urls: codeVideoResultUrls(fresh), text: formatCodeVideoCompleted(fresh) };
   547→  }
   548→  if (outcome.kind === "failed") {
   549→    const fresh = (await getTask(taskId))!;
   550→    const err = (fresh?.error_message ?? outcome.error ?? "").slice(0, 300);
   551→    return { kind: "failed", taskId, text: `❌ فشل رندر الفيديو البرمجي: ${err}` };
   552→  }
   553→  const fresh = (await getTask(taskId))!;
   554→  const stage = params.videoUrl ? "الرندر مكتمل" : params.audioUrl || !params.script ? "الرندر البرمجي" : "التعليق الصوتي";
   555→  return {
   556→    kind: "processing",
   557→    taskId,
   558→    progress: fresh?.progress ?? outcome.progress,
   559→    text:
   560→      `⏳ رندر الفيديو البرمجي يعمل — task_id: ${taskId}\n\n` +
   561→      `المرحلة: ${stage} — التقدم: ${fresh?.progress ?? outcome.progress}%\n\n` +
   562→      `تابع بـ get_task بنفس المعرّف كل ~10 ثوانٍ حتى الاكتمال (الكود والصوت والتوقيتات محفوظة ولا تُعاد من الصفر).`,
   563→  };
   564→}

   566→/** استكمال مهمة قائمة — أداة get_task ولوحة التحكم */
   567→export async function refreshCodeVideoTask(task: TaskRow, onProgress?: (progress: number, message: string) => Promise<void> | void): Promise<
   568→  | { kind: "completed"; taskId: string; urls: string[]; text: string }
   569→  | { kind: "processing"; taskId: string; progress: number; text: string }
   570→  | { kind: "failed"; taskId: string; text: string }
   571→  | { kind: "invalid"; text: string }
   572→> {
   573→  const params = parseParams(task);
   574→  if (!params) {
   575→    await updateTask(task.id, { status: "failed", error_message: "حالة الفيديو البرمجي تالفة في القاعدة" });
   576→    return { kind: "failed", taskId: task.id, text: "❌ حالة الفيديو البرمجي تالفة في قاعدة البيانات." };
   577→  }
   578→  if (task.status === "completed") {
   579→    return { kind: "completed", taskId: task.id, urls: codeVideoResultUrls(task), text: formatCodeVideoCompleted(task) };
   580→  }
   581→  const outcome = await advanceCodeVideo(task.id, params, PIPELINE_BUDGET_MS, onProgress ?? (() => {}));
   582→  if (outcome.kind === "completed") {
   583→    const fresh = (await getTask(task.id))!;
   584→    return { kind: "completed", taskId: task.id, urls: codeVideoResultUrls(fresh), text: formatCodeVideoCompleted(fresh) };
   585→  }
   586→  if (outcome.kind === "failed") {
   587→    const fresh = (await getTask(task.id))!;
   588→    const err = (fresh?.error_message ?? outcome.error ?? "").slice(0, 300);
   589→    return { kind: "failed", taskId: task.id, text: `❌ فشل رندر الفيديو البرمجي: ${err}` };
   590→  }
   591→  const fresh = (await getTask(task.id))!;
   592→  return {
   593→    kind: "processing",
   594→    taskId: task.id,
   595→    progress: fresh?.progress ?? outcome.progress,
   596→    text: `⏳ رندر الفيديو البرمجي قيد العمل — التقدم: ${fresh?.progress ?? outcome.progress}%\n\nتابع بـ get_task كل ~10 ثوانٍ.`,
   597→  };
   598→}
