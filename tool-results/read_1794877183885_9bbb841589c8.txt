   150→   150→    `${MERGER_BASE}/api/render-complete?op=provision`,
   151→   151→    {
   152→   152→      method: "POST",
   153→   153→      headers: { "Content-Type": "application/json" },

   155→   155→        html_code: String(payload.html_code || ""),
   156→   156→        title: String(payload.title || "فيديو برمجي — رندر كامل"),
   157→   157→        width: Number(process.env.INPUT_WIDTH) || Number(payload.width) || DEFAULT_WIDTH,
   158→   158→        height: Number(process.env.INPUT_HEIGHT) || Number(payload.height) || DEFAULT_HEIGHT,
   159→   159→        fps: Number(process.env.INPUT_FPS) || Number(payload.fps) || DEFAULT_FPS,

   163→   163→      }),
   164→   164→    },

   168→   168→  if (!res.ok) throw new Error(`فشل provision (HTTP ${res.status}): ${text.slice(0, 200)}`);

   190→   190→  throw new Error("لا payload_url ولا html_code_b64 — لا يمكن تحديد كود المشهد");

   196→   196→async function renderVideo({ html, width, height, fps, duration, wordTimings, audioPath }) {
   197→   197→  const ffmpeg = await ensureFfmpeg();
   198→   198→  const browser = await launchBrowser(width, height);
   199→   199→  const tmpDir = fs.mkdtempSync(path.join(os.tmpdir(), "codevid-"));
   200→   200→  const outputPath = path.join(tmpDir, "output.mp4");
   201→   201→  try {
   202→   202→    const page = await browser.newPage();
   203→   203→    page.setDefaultNavigationTimeout(60_000);
   204→   204→    await page.setContent(injectFonts(html), { waitUntil: "load" });
   205→   205→
   206→   206→    const hasRender = await page.evaluate(() => typeof window.renderAtTime === "function" || typeof window.__renderAtTime === "function");
   207→   207→    if (!hasRender) {
   208→   208→      throw new Error('صفحة HTML لا تعرّف window.renderAtTime(t, duration, wordTimings) — أضفها في <script> داخل الصفحة.');
   209→   209→    }
   210→   210→    if (wordTimings) {
   211→   211→      await page.evaluate((wt) => { window.__wordTimings = wt; }, wordTimings);
   212→   212→    }
   213→   213→    // نداء تسخين بالمدة الحقيقية كي لا يرمي renderAtTime خطأ عند التقسيم على المدة
   214→   214→    await page.evaluate((d) => {
   215→   215→      try {
   216→   216→        window.renderAtTime && window.renderAtTime(0, d, window.__wordTimings);
   217→   217→      } catch (e) {
   218→   218→        window.__mekoRenderError = String((e && e.message) || e);
   219→   219→      }
   220→   220→    }, duration);
   221→   221→    const warmError = await page.evaluate(() => window.__mekoRenderError || null);
   222→   222→    if (warmError) throw new Error(`renderAtTime ألقت خطأً عند t=0: ${warmError.slice(0, 300)}`);
   223→   223→
   224→   224→    // أمر ffmpeg — الإطارات تُضخ عبر stdin أثناء الالتقاط (بلا ملفات وسيطة)
   225→   225→    const args = ["-y", "-hide_banner", "-loglevel", "error", "-progress", "pipe:1", "-nostats"];
   226→   226→    args.push("-f", "image2pipe", "-framerate", String(fps), "-i", "-");
   227→   227→    if (audioPath) {
   228→   228→      args.push("-i", audioPath);
   229→   229→    } else {
   230→   230→      args.push("-f", "lavfi", "-t", duration.toFixed(3), "-i", "anullsrc=channel_layout=stereo:sample_rate=44100");
   231→   231→    }
   232→   232→    args.push("-map", "0:v:0", "-map", "1:a:0");
   233→   233→    // veryfast: ذاكرة أقل بكثير من medium بنفس الجودة البصرية تقريبًا عند CRF 18
   234→   234→    // (قابل للضبط بـ FFMPEG_X264_PRESET للبيئات الغنية بالذاكرة)
   235→   235→    args.push("-c:v", "libx264", "-crf", "18", "-preset", process.env.FFMPEG_X264_PRESET || "veryfast", "-pix_fmt", "yuv420p", "-r", String(fps));
   236→   236→    args.push("-c:a", "aac", "-b:a", "192k", "-ar", "44100");
   237→   237→    args.push("-t", duration.toFixed(3), "-movflags", "+faststart");
   238→   238→    args.push(outputPath);
   239→   239→
   240→   240→    const child = spawn(ffmpeg, args, { stdio: ["pipe", "ignore", "pipe"] });
   241→   241→    let ffmpegErr = "";
   242→   242→    child.stderr.on("data", (d) => {
   243→   243→      ffmpegErr += d.toString();
   244→   244→      if (ffmpegErr.length > 20000) ffmpegErr = ffmpegErr.slice(-20000);
   245→   245→    });
   246→   246→    const closed = new Promise((resolve, reject) => {
   247→   247→      child.on("error", reject);
   248→   248→      child.on("close", (code) => (code === 0 ? resolve() : reject(new Error(`ffmpeg خرج بكود ${code}: ${ffmpegErr.slice(-600)}`))));
   249→   249→    });
   250→   250→
   251→   251→    const frames = Math.round(fps * duration); // معدل ثابت حقيقي — بلا أي تخفيض
   252→   252→    let captured = 0;
   253→   253→    let lastPing = -1;
   254→   254→    for (let i = 0; i < frames; i++) {
   255→   255→      const t = (i * duration) / frames; // == i / fps بالضبط
   256→   256→      await page.evaluate((tt, d) => window.renderAtTime(tt, d, window.__wordTimings), t, duration);
   257→   257→      // PNG بلا فقد — جودة كاملة على عامل قوي بلا قيود زمن
   258→   258→      const buf = await page.screenshot({ type: "png", captureBeyondViewport: false });
   259→   259→      if (!child.stdin.write(buf)) {
   260→   260→        await new Promise((resolve) => child.stdin.once("drain", resolve));
   261→   261→      }
   262→   262→      captured++;
   263→   263→      // تقدم حي على المهمة (10..95) + ملخص الـ workflow كل ~10%
   264→   264→      const pct = Math.round(10 + 85 * ((i + 1) / frames));
   265→   265→      if (pct >= lastPing + 10 || i === frames - 1) {
   266→   266→        lastPing = pct;
   267→   267→        void pingProgress(Math.min(95, pct)).catch(() => {});
   268→   268→        log(`التقدم: ${captured}/${frames} إطارًا (${pct}%)`);
   269→   269→      }
   270→   270→    }
   271→   271→    child.stdin.end();
   272→   272→    await closed;
   273→   273→
   274→   274→    const st = fs.statSync(outputPath);
   275→   275→    if (st.size < 20_000) throw new Error(`الناتج صغير بشكل مريب (${st.size} بايت) — رندر فاشل غالبًا`);
   276→   276→    fs.copyFileSync(outputPath, PROGRESS_VIDEO_PATH);
   277→   277→    return { frames: captured, size: st.size };
   278→   278→  } finally {
   279→   279→    try { await browser.close(); } catch { /* تجاهل */ }
   280→   280→    fs.promises.rm(tmpDir, { recursive: true, force: true }).catch(() => {});
   281→   281→  }
   282→   282→}

   305→   305→    protocolTimeout: 120_000,
   306→   306→  });

   309→   309→/** كروم النظام على ubuntu-latest (google-chrome-stable مثبت مسبقًا)، وإن غاب يُثبت Chrome for Testing */
   310→   310→async function resolveChrome() {
   311→   311→  const candidates = ["/usr/bin/google-chrome-stable", "/usr/bin/google-chrome", "/usr/bin/chromium-browser", "/usr/bin/chromium"];
   312→   312→  for (const c of candidates) {
   313→   313→    if (fs.existsSync(c)) return c;
   314→   314→  }
   315→   315→  try {
   316→   316→    const found = execFileSync("which", ["google-chrome-stable"], { encoding: "utf8" }).trim();
   317→   317→    if (found) return found;
   318→   318→  } catch { /* نكمل للتثبيت */ }
   319→   319→  log("كروم النظام غير موجود — تثبيت Chrome for Testing…");

   335→   335→/** حقن خطي Meko (Noto Sans Arabic + Liberation Sans) كـ @font-face — نفس سلوك محرك Vercel */
   336→   336→function injectFonts(html) {
   337→   337→  const fontsDir = path.join(__dirname, "..", "merger", "fonts");
   338→   338→  const families = { MekoArabic: "NotoSansArabic-Regular.ttf", MekoLatin: "LiberationSans-Regular.ttf" };
   339→   339→  const faces = [];
   340→   340→  const chain = [];
   341→   341→  for (const [family, file] of Object.entries(families)) {
   342→   342→    try {
   343→   343→      const bytes = fs.readFileSync(path.join(fontsDir, file));
   344→   344→      if (bytes.length > 1000) {
   345→   345→        faces.push(`@font-face{font-family:'${family}';src:url('data:font/ttf;base64,${bytes.toString("base64")}') format('truetype');font-weight:normal;font-style:normal;font-display:block;}`);
   346→   346→        chain.push(`'${family}'`);
   347→   347→      }
   348→   348→    } catch { /* الخط غائب — نكمل بخطوط النظام */ }
   349→   349→  }
   350→   350→  const fallback = chain.length ? `html,body{font-family:${chain.join(",")},system-ui,-apple-system,'Segoe UI',sans-serif;}` : "";
   351→   351→  if (!faces.length && !fallback) return html;
   352→   352→  const tag = `<style data-meko-fonts>${faces.join("")}${fallback}</style>`;
   353→   353→  if (/<\/head>/i.test(html)) return html.replace(/<\/head>/i, `${tag}</head>`);
   354→   354→  if (/<body[^>]*>/i.test(html)) return html.replace(/<body[^>]*>/i, (m) => `${m}${tag}`);
   355→   355→  return tag + html;
   356→   356→}

   383→   383→      if (m) resolve(Number(m[1]) * 3600 + Number(m[2]) * 60 + Number(m[3]) + Number(m[4]) / 100);

   386→   386→  });

   407→   407→async function publishToMerger(meta) {
   408→   408→  try {
   409→   409→    const buf = fs.readFileSync(PROGRESS_VIDEO_PATH);
   410→   410→    const chunks = Math.ceil(buf.length / CHUNK_BYTES);
   411→   411→    for (let i = 0; i < chunks; i++) {
   412→   412→      const part = buf.subarray(i * CHUNK_BYTES, Math.min((i + 1) * CHUNK_BYTES, buf.length));
   413→   413→      const res = await fetchWithTimeout(
   414→   414→        `${MERGER_BASE}/api/render-complete?op=chunk&task_id=${TASK_ID}&job_id=${JOB_ID}&chunk=${i}`,
   415→   415→        { method: "POST", headers: { "Content-Type": "application/octet-stream" }, body: part },
   416→   416→        60_000,
   417→   417→      );
   418→   418→      if (!res.ok) throw new Error(`فشل رفع الجزء ${i + 1}/${chunks} (HTTP ${res.status}): ${(await res.text()).slice(0, 200)}`);
   419→   419→      log(`رفع الجزء ${i + 1}/${chunks} — ${(part.length / 1048576).toFixed(2)}MB`);
   420→   420→    }
   421→   421→    const done = await finalizeToMerger(meta);

   427→   427→    const contentType = videoRes.headers.get("content-type") || "";

   431→   431→    log(`تحقق النشر: صف المهمة HTTP ${jobRes.status} — الفيديو HTTP ${videoRes.status} مع content-type video`);

   433→   433→  } catch (e) {
   434→   434→    log(`فشل النشر عبر meko-manger: ${e instanceof Error ? e.message : String(e)}`);
   435→   435→    return null;
   436→   436→  }

   579→   579→function sanitizeTimings(raw) {
   580→   580→  if (!Array.isArray(raw) || raw.length < 2) return null;
   581→   581→  const list = raw.slice(0, MAX_WORD_TIMINGS).map((w) => {
   582→   582→    const o = w && typeof w === "object" ? w : {};
   583→   583→    const start = Number(o.start);
   584→   584→    const end = Number(o.end);
   585→   585→    const text = typeof o.text === "string" ? o.text.slice(0, 200) : "";
   586→   586→    if (!Number.isFinite(start) || !Number.isFinite(end) || end < start || !text) return null;
   587→   587→    return { start: Math.max(0, start), end: Math.max(0, end), text };
   588→   588→  }).filter(Boolean);
   589→   589→  if (list.length < 2) return null;
   590→   590→  list.sort((a, b) => a.start - b.start);
   591→   591→  return list;
   592→   592→}
